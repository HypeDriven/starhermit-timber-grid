# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com; never edit it). Re-run `npm run package` after every
# change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.16.0
engine.minor=14
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
control.up=ArrowUp | Move target up
control.down=ArrowDown | Move target down
control.left=ArrowLeft | Move target left
control.right=ArrowRight | Move target right
control.piece1=Digit1 | Select piece 1
control.piece2=Digit2 | Select piece 2
control.piece3=Digit3 | Select piece 3
control.next_piece=KeyN | Next piece
control.place=Enter+Space | Place piece
control.undo=KeyU | Undo
control.hint=KeyH | Hint
control.pause=KeyP+Escape | Pause / resume
control.camera=KeyC | Reset camera
engine.group.up=Board
engine.group.piece1=Pieces
engine.group.undo=Assists

# Gamepad defaults for the same actions.
engine.gamepad.up=DpadUp+LeftStickUp
engine.gamepad.down=DpadDown+LeftStickDown
engine.gamepad.left=DpadLeft+LeftStickLeft
engine.gamepad.right=DpadRight+LeftStickRight
engine.gamepad.piece1=LeftShoulder
engine.gamepad.piece3=RightShoulder
engine.gamepad.next_piece=East
engine.gamepad.place=South
engine.gamepad.undo=West
engine.gamepad.hint=North
